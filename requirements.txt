# Install with:  py -m pip install -r requirements.txt
#
# Deliberately minimal. Configuration parsing uses stdlib tomllib. By default
# the models are reached through the Codex and Claude CLIs that ship with
# their desktop apps, not an API client library.

python-docx>=1.2.0

# Timezone handling uses stdlib zoneinfo, but Windows (unlike Linux/macOS)
# ships no IANA tz database of its own — zoneinfo falls back to this package.
# Declared explicitly: it happened to already be installed (as some other
# package's transitive dependency) and PyInstaller's own hook silently
# bundled it, which is how this worked before it was ever listed here.
tzdata>=2024.1

# Auto-detects this PC's IANA timezone name for the first-run wizard's
# timezone dropdown (installer/first_run_wizard.py) — nothing in the stdlib
# maps "Windows timezone" to an IANA zone.
tzlocal>=5.0

# Stores API keys (provider_mode = "api") in the OS credential store (Windows
# Credential Manager here), never in config.toml. Required regardless of
# vendor. The anthropic/openai SDKs themselves are NOT listed here — they're
# lazily imported by paper_automation/providers/api_provider.py only when API
# mode is actually used, so an install that never turns it on carries none of
# that weight. Run `py -m pip install anthropic` / `openai` to use it.
keyring>=25.0.0

# Only needed if a Telegram/Slack/Teams notification channel is enabled.
requests>=2.31.0

# Web control panel. flask-login handles sessions for the admin/writer logins;
# waitress serves it properly once more than one person can reach it. Password
# hashing uses werkzeug.security, which arrives with Flask itself.
flask>=3.0.0
flask-login>=0.6.3
waitress>=3.0.0

# System tray icon (tray_app.py) for one-click Start/Stop and login autostart.
# Pillow draws the tray icon image; it's already pulled in transitively, but
# listed explicitly since tray_app.py imports it directly.
pystray>=0.19.5
Pillow>=10.0.0

# Only needed to run the test suite.
pytest>=8.0.0
